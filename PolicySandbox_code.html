// HeatGate Core Safety Ruleset v2.1
@id("baseline-permit")
permit(principal, action == Action::"Dispatch", resource);
@id("fair-rest-due")
@outcome("FORCE_REST")
@message("Mandatory heat rest cycle triggered")
forbid(principal, action == Action::"Dispatch", resource)
when {
context.heat_index >= 42 &&
context.active_stint_min >= 90
};
@id("peak-halt-override")
forbid(principal, action, resource) when { context.heat_index >= 48 };
